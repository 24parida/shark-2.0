cmake_minimum_required(VERSION 3.15)
project(shark_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
option(SHARK_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(SHARK_GUI_TESTS "Build GUI regression tests" OFF)
find_package(TBB REQUIRED)
set(BUILD_CARD5 OFF CACHE BOOL "" FORCE)
set(BUILD_CARD6 OFF CACHE BOOL "" FORCE)
set(BUILD_CARD7 OFF CACHE BOOL "" FORCE)
set(BUILD_PLO4 OFF CACHE BOOL "" FORCE)
set(BUILD_PLO5 OFF CACHE BOOL "" FORCE)
set(BUILD_PLO6 OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(../include/PokerHandEvaluator/cpp pheval)
file(GLOB CORE_SOURCES ../src/hands/*.cpp ../src/tree/*.cpp ../src/solver/*.cpp ../src/trainer/*.cpp)
add_library(shark_core STATIC ${CORE_SOURCES})
target_include_directories(shark_core PUBLIC ../src ../include/PokerHandEvaluator/cpp/include/phevaluator)
target_link_libraries(shark_core PUBLIC TBB::tbb pheval)
if(SHARK_SANITIZE)
  target_compile_options(shark_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer -D_GLIBCXX_ASSERTIONS)
  target_link_options(shark_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(solver_tests solver_tests.cpp)
target_link_libraries(solver_tests PRIVATE shark_core)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(solver_tests PRIVATE -fno-fast-math)
endif()
add_executable(reference_values reference_values.cpp)
target_link_libraries(reference_values PRIVATE shark_core)
foreach(suite normalization chance isomorphism betting keys ranges convergence response trained_isomorphism range_roundtrip tiny_weights)
  add_test(NAME ${suite} COMMAND solver_tests ${suite})
endforeach()
if(SHARK_GUI_TESTS)
  find_package(FLTK 1.4 CONFIG REQUIRED)
  file(GLOB GUI_SOURCES ../gui/components/*.cpp)
  add_executable(gui_tests gui_tests.cpp ${GUI_SOURCES})
  target_compile_options(gui_tests PRIVATE -fno-access-control)
  target_include_directories(gui_tests PRIVATE ../gui ${FLTK_INCLUDE_DIR})
  target_link_libraries(gui_tests PRIVATE shark_core fltk::fltk fltk::images)
  foreach(suite import weights navigation board cache average export compound_swaps edit_weights blocker_average)
    add_test(NAME gui_${suite} COMMAND xvfb-run -a $<TARGET_FILE:gui_tests> ${suite})
  endforeach()
endif()
